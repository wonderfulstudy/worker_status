// 定义配置文件路径变量
var configFile string

// 构建根 command 命令。前面我们介绍它还可以有子命令，这个command里没有构建子命令
var rootCmd = &cobra.Command{
	Use:   "nyx",
	Short: "简短介绍：查询nyx项目链上数据的工具",
	Long: `详细介绍：
		1、目前实现根据地址查询在线情况
		2、根据机器列表，将链上数据对应机器`
}

// 执行 rootCmd 命令并检测错误
func Execute() {
	cobra.CheckErr(rootCmd.Execute())
}

func init() {
    // 加载运行初始化配置
	cobra.OnInitialize(initConfig)
    // rootCmd，命令行下读取配置文件，持久化的 flag，全局的配置文件
	rootCmd.PersistentFlags().StringVar(&configFile, "config", "", "config file (default is config.yml)")
	// local flag，本地化的配置
	rootCmd.Flags().BoolP("toggle", "t", false, "Help message for toggle")
}

// 初始化配置的一些设置
func initConfig() {
	if configFile != "" {
		viper.SetConfigFile(configFile) // viper 设置配置文件
	} else {// 上面没有指定配置文件，下面就读取 home 下的 .firstappname.yaml文件
        // 配置文件参数设置
		home, err := os.UserHomeDir()
		cobra.CheckErr(err)

		viper.AddConfigPath(home)
		viper.SetConfigType("yaml")
		viper.SetConfigName(".firstappname")
	}

	viper.AutomaticEnv() 

	if err := viper.ReadInConfig(); err == nil {// 读取配置文件
		fmt.Fprintln(os.Stderr, "Using config file:", viper.ConfigFileUsed())
	}
}

	Addresses []string `yaml:"Addresses"`

// 全局配置变量
var GlobalConfig Config

func InitConfig() error {
	// 定义命令行参数
	configPath := flag.String("config", "config.yml", "path to config file")
	flag.Parse()

	// 读取配置文件
	config, err := readConfig(*configPath)
	if err != nil {
		return fmt.Errorf("error reading config file: %v", err)
	}

	// 设置全局配置
	GlobalConfig = *config

	fmt.Println("Addresses from config:", GlobalConfig.Addresses)
	return nil
}

func readConfig(filename string) (*Config, error) {
	data, err := ioutil.ReadFile(filename)
	if err != nil {
		return nil, err
	}

	var config Config
	err = yaml.Unmarshal(data, &config)
	if err != nil {
		return nil, err
	}

	return &config, nil
}

// 读取链上数据
func get_registered_workers() int {
	url := "https://dev-squid.cybros.network/graphql"
	data := []byte(`
		{
			"operationName": "MyQuery",
			"query": "query MyQuery {\n  workers(where: {ownerAddress_eq: \"5ES6Mu3uUUB8QRarGB4J73RnoeTxADWV8HKhnKfrASpoLSEP\"}) {\n    status\n    address\n  }\n}\n",
			"variables":null
		}`)

	// 发起 POST 请求
	resp, post_err := http.Post(url, "application/json", bytes.NewBuffer(data))
	if post_err != nil {
		fmt.Println("发起请求失败:", post_err)
		return -1
	}
	defer resp.Body.Close()

	// 读取响应
	body, resp_err := ioutil.ReadAll(resp.Body)
	if resp_err != nil {
		fmt.Println("读取响应失败:", resp_err)
		return -1
	}

	// 定义目标结构体变量
	var response Response
	// 解析 JSON 数据
	json_err := json.Unmarshal([]byte(string(body)), &response)
	if json_err != nil {
		fmt.Println("解析 JSON 失败:", json_err)
		return -1
	}

	// 打印解析结果
	count := 0
	fmt.Println("解析成功，workers 信息如下：")
	for _, worker := range response.Data.Workers {
		fmt.Printf("Status: %s, Address: %s\n", worker.Status, worker.Address)
		count++
	}

	return count
}

func main() {
	// 初始化配置
	err := InitConfig()
	if err != nil {
		fmt.Printf("Failed to initialize config: %v\n", err)
		return
	}

	fmt.Print("%s", GlobalConfig.Addresses)

	// 创建一个新的注册表
	registry := prometheus.NewRegistry()

	// 注册机器总数指标
	registeredWorkers := prometheus.NewGauge(prometheus.GaugeOpts{
		Name: "registeredWorkers", // 指标名称
		Help: "在链上已经注册的机器总数",      // 指标帮助信息
	})

	// 注册指标
	registry.MustRegister(registeredWorkers)

	// 增加指标值
	go func() {
		for {
			count := get_registered_workers()
			registeredWorkers.Set(float64(count))
			time.Sleep(10 * time.Second)
		}
	}()

	// 创建一个HTTP处理器来暴露指标
	http.Handle("/metrics", promhttp.HandlerFor(registry, promhttp.HandlerOpts{}))

	// 启动Web服务器
	http.ListenAndServe(":8080", nil)
}
